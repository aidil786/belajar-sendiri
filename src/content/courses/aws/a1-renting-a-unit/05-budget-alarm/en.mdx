<Interactive name="SortCards" buckets={[{ id: 'zero', label: 'Zero spend budget' }, { id: 'monthly', label: 'Monthly cost budget' }]} items={[{ label: 'Emails once you spend past the free limits', bucket: 'zero', why: 'Any real spend at all sets it off.' }, { label: 'You pick a monthly amount', bucket: 'monthly', why: 'You choose the line yourself.' }, { label: 'Warns when the forecast will pass your line', bucket: 'monthly', why: 'It looks ahead, not only back.' }, { label: 'A great first alarm for a learner', bucket: 'zero', why: 'Learners usually plan to spend nothing.' }]} success="Use both: zero spend first, a monthly line as a backstop." />
