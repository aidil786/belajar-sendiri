<Interactive name="Mission" id="aws-m10-build" steps={[
  { id: 'repo', text: 'Buat rak, dengan imbasan semasa push.', command: 'aws ecr create-repository --repository-name kedai-kancil --image-scanning-configuration scanOnPush=true' },
  { id: 'push', text: 'Log masuk (pelajaran A4-1), kemudian tag dan push image Kedai Kancil dari Dunia Docker D4.', command: 'docker push ACCOUNT_ID.dkr.ecr.REGION.amazonaws.com/kedai-kancil:1.0' },
  { id: 'cluster', text: 'Buat kluster. ECS sendiri percuma.', command: 'aws ecs create-cluster --cluster-name kedai' },
  { id: 'card', text: 'Simpan taskdef.json: cpu 256, memory 512, image anda, port 80, ecsTaskExecutionRole. Daftarkan.', command: 'aws ecs register-task-definition --cli-input-json file://taskdef.json' },
  { id: 'door', text: 'Buat pengawal pintu dalam VPC lalai. Benarkan port 80 dari YOUR_IP/32 sahaja.', command: 'aws ec2 authorize-security-group-ingress --group-id SG_ID --protocol tcp --port 80 --cidr YOUR_IP/32' },
  { id: 'service', text: 'Mulakan service dengan satu task dan IP awam.', command: 'aws ecs create-service --cluster kedai --service-name shop --task-definition kedai-web --desired-count 1 --launch-type FARGATE --network-configuration "awsvpcConfiguration={subnets=[SUBNET_ID],securityGroups=[SG_ID],assignPublicIp=ENABLED}"' },
  { id: 'visit', text: 'Cari IP awam task itu dalam konsol ECS. Buka dalam pelayar anda.' }]}
