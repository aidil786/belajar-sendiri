<Card id="try-run" type="try">

<Interactive name="Choice" options={[{ label: 'Tidak: fork tak dapat secret, jadi tiada token', correct: true, why: 'GitHub jauhkan secret anda daripada kod orang luar.' }, { label: 'Ya, sama macam anda', why: 'Pull request dari fork berjalan tanpa secret anda.' }, { label: 'Hanya kalau pull request itu kecil', why: 'Saiz tak penting.
