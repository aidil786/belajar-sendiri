<Card id="try-build" type="try">
Room 2: build the safe PR helper. Each decoy is one of the holes.

<Interactive name="YamlBuilder" workflow="PR helper" goal={{ order: ['pr', 'read', 'job', 'checkout', 'pin', 'title', 'test'] }} blocks={[
  { id: 'prt', kind: 'trigger', label: 'On pull_request_target', yaml: 'pull_request_target:', means: 'Run with your secrets and a write key, even for forks.', why: "It hands your keys to the stranger's code. Use pull_request." },
  { id: 'pr', kind: 'trigger', label: 'On pull_request', yaml: 'pull_request:', means: 'Fork PRs get no secrets and a read-only key.' },
  { id: 'read', kind: 'permissions', label: 'Read code only', yaml: 'contents: read', means: 'The key can only read your code.' },
  { id: 'lots', kind: 'permissions', label: 'Write lots', yaml: 'contents: write\npull-requests: write\npackages: write', means: 'The key can change code, PRs and packages.', why: 'This job only reads and tests. Read is enough.' },
  { id: 'self', kind: 'job', label: 'Job on your office PC', yaml: 'check:\n  runs-on: self-hosted', means: 'Run on your own machine.', why: "On a public repo, a stranger's PR could run on your machine." },
  { id: 'job', kind: 'job', label: 'Job on a hosted runner', yaml: 'check:\n  runs-on: ubuntu-latest', means: 'A fresh machine, wiped after the job.' },
  { id: 'checkout', kind: 'step', label: 'Check out code', yaml: '- uses: actions/checkout@v7\n  with:\n    persist-credentials: false', means: 'Copy the code. Keep the key out of Git config.' },
  { id: 'tag', kind: 'step', label: 'Size check (tag)', yaml: '- uses: example-org/size-check@v3', means: 'Borrow the size check by its tag.', why: 'A tag can be moved. Pin the full SHA.' },

  { id: 'inline', kind: 'step', label: 'Print the title (pasted in)', yaml: '- run: echo "Checking ${{ github.event.pull_request.title }}"', means: 'Paste the title into the script.', why: 'A sneaky title becomes a command. Pass it through env.' },
  { id: 'title', kind: 'step', label: 'Print the title (in env)', yaml: '- run: echo "Checking $TITLE"\n  env:\n    TITLE: ${{ github.event.pull_request.title }}', means: 'The title arrives as plain data.' },
