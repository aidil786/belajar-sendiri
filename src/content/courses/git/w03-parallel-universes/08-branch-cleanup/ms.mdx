<Card id="check-sort" type="check">

<Interactive name="SortCards" buckets={[{ id: 'ok', label: 'Dipadam' }, { id: 'no', label: 'Git kata tidak' }]} items={[{ label: 'about, dah di-merge ke main', bucket: 'ok', why: 'Kerjanya selamat dalam main.' }, { label: 'dark-mode, dah di-merge ke main', bucket: 'ok', why: 'Kerjanya selamat dalam main.' }, { label: 'Branch dengan kerja yang belum di-merge', bucket: 'no', why: 'Guna -D hanya kalau anda pasti.' }, { label: 'main, semasa anda berada di situ', bucket: 'no', why: 'Anda tak boleh padam branch tempat anda berada.' }]} success="Dah susun! -d jaga kerja anda supaya selamat." />
