<Interactive name="OrderList" items={['The alert rings at 3 a.m.', 'Check the dashboard: errors are up', 'Roll back the last release', 'Orders work again', 'Write down what happened, with no blame', 'Automate a fix so it never repeats']} label="First thing at the top" success="That's an SRE's night: fix it fast, then fix it for good." />
