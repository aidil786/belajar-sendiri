<Interactive name="PromQLPlayground" scenario="o2-shop-requests" initial={'rate(http_requests_total{job="shop"}[5m])'} goal={{ resultCount: 1, labelNames: [], usesFunction: ['sum', 'rate'], typed: 'job="shop"' }} hints={['Balut semuanya dengan sum( … ).']} solution={'sum(rate(http_requests_total{job="shop"}[5m]))'} success="Satu nombor untuk satu panel Stat yang besar." />
