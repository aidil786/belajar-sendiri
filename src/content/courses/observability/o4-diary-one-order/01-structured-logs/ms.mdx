<Interactive name="Choice" compact options={[{ label: 'level', correct: true, why: 'Betul. level="error" jumpa setiap ralat, apa pun bunyi msg.' }, { label: 'msg', why: 'Mesej boleh tukar ayat. <Card id="deeper" type="deeper">
